        function updateDateStamp() {
            const options = { year: 'numeric', month: 'long', day: 'numeric' };
            const today = new Date().toLocaleDateString('id-ID', options);
            const footerEl = document.getElementById('last-updated');

        // Menampilkan notifikasi toast halus
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            
            const bgColor = type === 'success' ? 'bg-emerald-600' : 'bg-slate-800';
            toast.className = `${bgColor} text-white px-4 py-3 rounded-xl shadow-lg text-sm font-medium flex items-center gap-2 transform transition-all duration-300 translate-y-2 opacity-0`;
            toast.innerHTML = `<span>${message}</span>`;
            
            container.appendChild(toast);
            
            // Animasi masuk
            requestAnimationFrame(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            });

            // Hapus setelah 3 detik
            setTimeout(() => {
                toast.classList.add('translate-y-2', 'opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }

        // Mengubah Tema Warna
        function setTheme(themeName) {
            const htmlEl = document.documentElement;
            if (themeName === 'blue') {
                htmlEl.removeAttribute('data-theme');
            } else {
                htmlEl.setAttribute('data-theme', themeName);
            }

            // Update indikator ring pada tombol tema
            ['blue', 'emerald', 'slate'].forEach(t => {
                const btn = document.getElementById(`btn-theme-${t}`);
                if (t === themeName) {
                    btn.classList.add('ring-2');
                } else {
                    btn.classList.remove('ring-2');
                }
            });

            const names = { blue: 'Biru Profesional', emerald: 'Hijau Emerald', slate: 'Abu-abu Minimalis' };
            showToast(`Tema diubah ke ${names[themeName]}`);
        }

        // Mengaktifkan / Menonaktifkan Mode Edit
        function toggleEditMode() {
            isEditMode = !isEditMode;
            const doc = document.getElementById('biodata-document');
            const btnText = document.getElementById('text-toggle-edit');
            const btnToggle = document.getElementById('btn-toggle-edit');
            const editBanner = document.getElementById('edit-banner');
            const editableFields = document.querySelectorAll('.editable-field');
            const editControls = document.querySelectorAll('.edit-only-control');
            const photoLabel = document.getElementById('upload-photo-label');

            if (isEditMode) {
                doc.classList.add('editing-active');
                btnText.textContent = 'Selesai Edit';
                btnToggle.classList.replace('bg-theme-primary', 'bg-emerald-600');
                editBanner.classList.remove('hidden');
                
                editableFields.forEach(el => {
                    el.setAttribute('contenteditable', 'true');
                });
                
                editControls.forEach(el => {
                    el.classList.remove('hidden');
                });

                // Tampilkan indikator ganti foto lebih jelas saat mode edit
                photoLabel.classList.remove('opacity-0');
                photoLabel.classList.add('opacity-90');

                showToast('Mode Edit aktif. Silakan klik teks untuk mengedit.');
            } else {
                doc.classList.remove('editing-active');
                btnText.textContent = 'Edit Biodata';
                btnToggle.classList.replace('bg-emerald-600', 'bg-theme-primary');
                editBanner.classList.add('hidden');
                
                editableFields.forEach(el => {
                    el.removeAttribute('contenteditable');
                });
                
                editControls.forEach(el => {
                    el.classList.add('hidden');
                });

                photoLabel.classList.add('opacity-0');
                photoLabel.classList.remove('opacity-90');

                showToast('Perubahan disimpan di halaman.', 'success');
            }
        }

        // Menangani Upload Foto Profil
        function handleImageUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            if (!file.type.startsWith('image/')) {
                showToast('Harap pilih file gambar (JPG, PNG, WEBP).');
                return;
            }

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = document.getElementById('profile-image');
                img.src = e.target.result;
                showToast('Foto profil berhasil diperbarui!', 'success');
            };
            reader.readAsDataURL(file);
        }

        // Menambah Tag Keahlian Baru
        function addSkillBadge() {
            const input = document.getElementById('new-skill-input');
            const val = input.value.trim();
            if (!val) {
                showToast('Ketik nama keahlian terlebih dahulu.');
                return;
            }

            const container = document.getElementById('skills-container');
            const badge = document.createElement('span');
            badge.className = 'skill-badge inline-flex items-center gap-1.5 px-3.5 py-1.5 rounded-xl bg-theme-light text-theme border border-theme text-sm font-semibold';
            
            const editableAttr = isEditMode ? 'contenteditable="true"' : '';
            const hiddenClass = isEditMode ? '' : 'hidden';

            // Mencegah injeksi HTML dengan textContent
            const textSpan = document.createElement('span');
            textSpan.className = 'editable-field';
            if (isEditMode) textSpan.setAttribute('contenteditable', 'true');
            textSpan.textContent = val;

            const delBtn = document.createElement('button');
            delBtn.type = 'button';
            delBtn.title = 'Hapus keahlian';
            delBtn.className = `edit-only-control ${hiddenClass} no-print text-rose-500 hover:text-rose-700 ml-1 cursor-pointer`;
            delBtn.onclick = function() { removeSkillBadge(this); };
            delBtn.innerHTML = '<i data-lucide="x" class="w-3.5 h-3.5"></i>';

            badge.appendChild(textSpan);
            badge.appendChild(delBtn);
            container.appendChild(badge);
            
            input.value = '';
            lucide.createIcons();
            showToast(`Keahlian "${val}" ditambahkan.`);
        }

        // Mengunduh File HTML yang Sudah Diisi Pengguna
        function downloadHTML() {
            // Pastikan mode edit dimatikan agar file yang diunduh bersih
            const wasEditing = isEditMode;
            if (wasEditing) {
                toggleEditMode();
            }

            // Ambil seluruh kode HTML dokumen saat ini
            const fullHTML = '<!DOCTYPE html>\n' + document.documentElement.outerHTML;
            const blob = new Blob([fullHTML], { type: 'text/html;charset=utf-8' });
            const url = URL.createObjectURL(blob);

            // Buat nama file berdasarkan nama lengkap yang diisi
            let fileName = document.getElementById('data-fullname').innerText.trim();
            if (!fileName || fileName === '[Isi nama lengkap]') {
                fileName = 'biodata-diri';
            } else {
                fileName = 'biodata-' + fileName.toLowerCase().replace(/[^a-z0-9]+/g, '-');
            }

            const a = document.createElement('a');
            a.href = url;
            a.download = `${fileName}.html`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);

            showToast(`File ${fileName}.html berhasil diunduh!`, 'success');
        }
